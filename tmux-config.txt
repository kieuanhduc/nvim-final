# Tmux Configuration

#.tmux.config

# Based on: https://github.com/josean-dev/dev-environment-files

#--------------------------------------------------------------------------
# General Settings
#--------------------------------------------------------------------------

# Set true color
set-option -sa terminal-overrides ",xterm*:Tc"

# Enable mouse support
set -g mouse on

# Enable clipboard với OSC 52 (for SSH/remote)
set -g set-clipboard on
set -as terminal-features ',*:clipboard'
set -s set-clipboard external

# Start windows and panes at 1, not 0
set -g base-index 1
set -g pane-base-index 1
set-window-option -g pane-base-index 1
set-option -g renumber-windows on

# Set prefix
unbind C-b
set -g prefix C-a
bind C-a send-prefix

#--------------------------------------------------------------------------
# Key Bindings
#--------------------------------------------------------------------------

# Vim style pane selection
bind h select-pane -L
bind j select-pane -D
bind k select-pane -U
bind l select-pane -R

# Use Alt-arrow keys without prefix key to switch panes
bind -n M-Left select-pane -L
bind -n M-Right select-pane -R
bind -n M-Up select-pane -U
bind -n M-Down select-pane -D

# Shift arrow to switch windows
bind -n S-Left  previous-window
bind -n S-Right next-window

# Shift Alt vim keys to switch windows
bind -n M-H previous-window
bind -n M-L next-window

# Split panes using | and -
bind | split-window -h -c "#{pane_current_path}"
bind - split-window -v -c "#{pane_current_path}"
unbind '"'
unbind %

# Reload config
bind r source-file ~/.tmux.conf \; display "Config Reloaded!"

# Open new windows in current path
bind c new-window -c "#{pane_current_path}"

#--------------------------------------------------------------------------
# Copy Mode (Vi style)
#--------------------------------------------------------------------------

# Set vi-mode
set-window-option -g mode-keys vi

# Keybindings for copy mode với OSC 52 (works over SSH!)
bind-key -T copy-mode-vi v send-keys -X begin-selection
bind-key -T copy-mode-vi C-v send-keys -X rectangle-toggle
bind-key -T copy-mode-vi y send-keys -X copy-selection-and-cancel

# Mouse drag to copy (OSC 52)
bind-key -T copy-mode-vi MouseDragEnd1Pane send-keys -X copy-selection-and-cancel

#--------------------------------------------------------------------------
# Theme & Colors (Tokyo Night)
#--------------------------------------------------------------------------

# Status bar colors
set -g status-style 'bg=#1a1b26 fg=#c0caf5'

# Window status
setw -g window-status-current-style 'fg=#1a1b26 bg=#7aa2f7 bold'
setw -g window-status-current-format ' #I#[fg=#1a1b26]:#[fg=#1a1b26]#W#[fg=#1a1b26]#F '

setw -g window-status-style 'fg=#7aa2f7 bg=#1a1b26'
setw -g window-status-format ' #I#[fg=#7aa2f7]:#[fg=#7aa2f7]#W#[fg=#7aa2f7]#F '

# Pane borders
set -g pane-border-style 'fg=#1d202f'
set -g pane-active-border-style 'fg=#7aa2f7'

# Message text
set -g message-style 'fg=#c0caf5 bg=#1a1b26 bold'

#--------------------------------------------------------------------------
# Status Bar
#--------------------------------------------------------------------------

set -g status-position top
set -g status-interval 1
set -g status-justify left
set -g status-left-length 200
set -g status-right-length 200

# Left: Session name
set -g status-left '#[fg=#1a1b26,bg=#7aa2f7,bold] #S #[fg=#7aa2f7,bg=#1a1b26]'

# Right: Date and time
set -g status-right '#[fg=#7aa2f7,bg=#1a1b26]#[fg=#1a1b26,bg=#7aa2f7,bold] %Y-%m-%d %H:%M '

#--------------------------------------------------------------------------
# Plugins (tpm - Tmux Plugin Manager)
#--------------------------------------------------------------------------

# List of plugins
set -g @plugin 'tmux-plugins/tpm'
set -g @plugin 'tmux-plugins/tmux-sensible'
set -g @plugin 'christoomey/vim-tmux-navigator'
set -g @plugin 'tmux-plugins/tmux-yank'

# Tokyo Night theme for tmux (optional)
# set -g @plugin "janoamaral/tokyo-night-tmux"

#--------------------------------------------------------------------------
# Plugin Settings
#--------------------------------------------------------------------------

# tmux-yank settings
set -g @yank_selection_mouse 'clipboard'

#--------------------------------------------------------------------------
# Initialize TMUX plugin manager (keep this line at the very bottom)
#--------------------------------------------------------------------------

# Install tpm if not installed
if "test ! -d ~/.tmux/plugins/tpm" \
   "run 'git clone https://github.com/tmux-plugins/tpm ~/.tmux/plugins/tpm && ~/.tmux/plugins/tpm/bin/install_plugins'"

run '~/.tmux/plugins/tpm/bin/tpm'
